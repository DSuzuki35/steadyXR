cmake_minimum_required(VERSION 3.25)
project(SteadyXR_OBS VERSION 1.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SteadyXR OBS requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h" OR NOT EXISTS "${OBS_IMPORT_LIB}")
  message(FATAL_ERROR "Set OBS_SOURCE_DIR and OBS_IMPORT_LIB, or run Build.ps1")
endif()
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h")
add_library(obs-runtime SHARED IMPORTED)
set_target_properties(obs-runtime PROPERTIES IMPORTED_IMPLIB "${OBS_IMPORT_LIB}"
  IMPORTED_LOCATION "${OBS_ROOT}/bin/64bit/obs.dll"
  INTERFACE_INCLUDE_DIRECTORIES "${OBS_SOURCE_DIR}/libobs;${CMAKE_CURRENT_BINARY_DIR}")
add_library(steadyxr-obs MODULE source.cpp version.rc)
target_link_libraries(steadyxr-obs PRIVATE obs-runtime d3d11 dxgi)
target_compile_definitions(steadyxr-obs PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(steadyxr-obs PRIVATE /W4 /utf-8)
install(TARGETS steadyxr-obs LIBRARY DESTINATION bin/64bit RUNTIME DESTINATION bin/64bit)
install(DIRECTORY data/ DESTINATION data)
install(FILES Install-OBS-Integration.ps1 README.md LICENSE DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(steadyxr-lifecycle-fixture WIN32 tests/lifecycle-fixture.cpp)
  set_target_properties(steadyxr-lifecycle-fixture PROPERTIES OUTPUT_NAME SteadyXR
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixture")
  add_executable(steadyxr-lifecycle-test tests/lifecycle.cpp)
  foreach(target steadyxr-lifecycle-fixture steadyxr-lifecycle-test)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  endforeach()
  add_test(NAME steadyxr-lifecycle COMMAND steadyxr-lifecycle-test "$<TARGET_FILE:steadyxr-lifecycle-fixture>")
  set_tests_properties(steadyxr-lifecycle PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
  find_program(POWERSHELL_EXECUTABLE NAMES powershell REQUIRED)
  add_test(NAME steadyxr-installer COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/installer.ps1" -SourceDirectory "${CMAKE_CURRENT_SOURCE_DIR}"
    -PluginDll "$<TARGET_FILE:steadyxr-obs>" -FixtureExe "$<TARGET_FILE:steadyxr-lifecycle-fixture>"
    -ObsRoot "${OBS_ROOT}" -BuildDirectory "${CMAKE_CURRENT_BINARY_DIR}")
  set_tests_properties(steadyxr-installer PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  add_executable(steadyxr-obs-test tests/integration.cpp)
  target_link_libraries(steadyxr-obs-test PRIVATE obs-runtime d3d11 dxgi)
  target_compile_definitions(steadyxr-obs-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(steadyxr-obs-test PRIVATE /W4 /utf-8)
  add_test(NAME steadyxr-obs-gpu COMMAND steadyxr-obs-test "${OBS_ROOT}"
    "$<TARGET_FILE:steadyxr-obs>" "${CMAKE_CURRENT_SOURCE_DIR}/data")
  set_tests_properties(steadyxr-obs-gpu PROPERTIES
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${OBS_ROOT}/bin/64bit" TIMEOUT 60 RUN_SERIAL TRUE)
endif()
