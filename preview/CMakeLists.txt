# One build-time invocation supplies the same stamp to the application and layer.
# Unlike configure_file timestamps, this refreshes when an existing tree is rebuilt.
add_custom_target(vrst-preview-build-stamp
  COMMAND "${CMAKE_COMMAND}" "-DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}"
          "-DPREVIEW_VERSION=${PROJECT_VERSION}" -P "${CMAKE_CURRENT_SOURCE_DIR}/generate_build_stamp.cmake"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/generated/steadyxr_build_stamp.hpp"
             "${CMAKE_CURRENT_BINARY_DIR}/steadyxr-build.json"
  COMMENT "Stamping the SteadyXR preview build (no expiry)"
  VERBATIM)
add_library(vrst-preview-build-expiry INTERFACE)
