if(NOT WIN32)
  message(FATAL_ERROR "Standalone preview currently requires Windows x64")
endif()
if(NOT VRST_OPENXR_DIR AND EXISTS "${PROJECT_SOURCE_DIR}/third_party/openxr/include/openxr/openxr.h")
  set(VRST_OPENXR_DIR "${PROJECT_SOURCE_DIR}/third_party/openxr")
endif()
if(NOT EXISTS "${VRST_OPENXR_DIR}/include/openxr/openxr_loader_negotiation.h")
  message(FATAL_ERROR "VRST_OPENXR_DIR must point to OpenXR SDK headers (see scripts/prepare-deps.py)")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Standalone preview requires a 64-bit build")
endif()
add_library(vrst-preview-transport STATIC transport.cpp)
target_include_directories(vrst-preview-transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(vrst-preview-transport PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(horizonlock-openxr-layer SHARED layer.cpp)
target_include_directories(horizonlock-openxr-layer PRIVATE "${VRST_OPENXR_DIR}/include")
target_link_libraries(horizonlock-openxr-layer PRIVATE vrst-preview-transport d3d11 dxgi)
set_target_properties(horizonlock-openxr-layer PROPERTIES PREFIX "")
configure_file(openxr-layer.json.in XR_APILAYER_HORIZONLOCK_preview.json COPYONLY)
add_library(vrst-preview-renderer STATIC renderer.cpp composition.cpp)
target_link_libraries(vrst-preview-renderer PUBLIC vrst-preview-transport vrst_core d3d11 dxgi d3dcompiler user32 gdi32)
add_executable(horizonlock-preview WIN32 main.cpp)
add_library(vrst-window-capture STATIC window_capture.cpp)
target_compile_definitions(vrst-window-capture PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(vrst-window-capture PUBLIC d3d11 dxgi dwmapi runtimeobject user32)
target_link_libraries(horizonlock-preview PRIVATE vrst-preview-renderer vrst-window-capture comctl32 shell32 advapi32 winmm)
if(MINGW)
  target_link_options(horizonlock-preview PRIVATE -municode)
endif()
install(TARGETS horizonlock-preview horizonlock-openxr-layer RUNTIME DESTINATION preview)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/XR_APILAYER_HORIZONLOCK_preview.json" README-DE.md DESTINATION preview)
