cmake_minimum_required(VERSION 3.24)
project(vr_stabilizer VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  # Keep the standalone sender usable without a separate VC++ redistributable.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(VRST_BUILD_PLUGIN "Build OBS filter" OFF)
option(VRST_BUILD_SENDER "Build SteamVR pose sender" OFF)
option(VRST_BUILD_TESTS "Build portable tests" ON)
if(VRST_BUILD_TESTS)
  enable_testing()
endif()
option(VRST_BUILD_PREVIEW "Build standalone OpenXR preview for Windows" OFF)
set(VRST_OPENXR_DIR "" CACHE PATH "Khronos OpenXR SDK root")
set(VRST_OBS_SOURCE_DIR "" CACHE PATH "OBS source root (libobs headers)")
set(VRST_OBS_LIBRARY "" CACHE FILEPATH "OBS import library or native libobs")
set(VRST_OBS_GENERATED_DIR "" CACHE PATH "Directory containing generated obsconfig.h")
set(VRST_SIMDE_DIR "" CACHE PATH "SIMDe root for native ARM verification")
set(VRST_OPENVR_DIR "" CACHE PATH "Valve OpenVR SDK root")
add_library(vrst_core STATIC src/core.cpp src/protocol.cpp src/image.cpp src/advanced.cpp src/calibration.cpp src/timing_overlay.cpp src/spectator.cpp)
target_include_directories(vrst_core PUBLIC include)
find_package(Threads REQUIRED)
add_library(vrst_receiver STATIC src/receiver.cpp)
target_link_libraries(vrst_receiver PUBLIC vrst_core Threads::Threads)
if(WIN32)
  target_link_libraries(vrst_receiver PUBLIC ws2_32)
endif()
if(VRST_BUILD_PREVIEW)
  add_subdirectory(preview)
endif()
if(VRST_BUILD_PLUGIN)
  if(VRST_OBS_SOURCE_DIR AND VRST_OBS_LIBRARY)
    add_library(OBS::libobs UNKNOWN IMPORTED)
    set_target_properties(OBS::libobs PROPERTIES IMPORTED_LOCATION "${VRST_OBS_LIBRARY}")
    set_property(TARGET OBS::libobs PROPERTY INTERFACE_INCLUDE_DIRECTORIES
      "${VRST_OBS_SOURCE_DIR}/libobs;${VRST_OBS_GENERATED_DIR}")
    if(VRST_SIMDE_DIR)
      set_property(TARGET OBS::libobs APPEND PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${VRST_SIMDE_DIR}")
    endif()
  else()
    find_package(libobs REQUIRED)
  endif()
  add_subdirectory(plugin)
endif()
if(VRST_BUILD_SENDER)
  add_executable(vrst-sender sender/main.cpp sender/overlay.cpp)
  target_link_libraries(vrst-sender PRIVATE vrst_core Threads::Threads)
  if(WIN32)
    if(NOT VRST_OPENVR_DIR)
      message(FATAL_ERROR "VRST_OPENVR_DIR is required for the Windows sender")
    endif()
    target_compile_definitions(vrst-sender PRIVATE VRST_WITH_OPENVR=1)
    target_include_directories(vrst-sender SYSTEM PRIVATE "${VRST_OPENVR_DIR}/headers")
    target_link_libraries(vrst-sender PRIVATE "${VRST_OPENVR_DIR}/lib/win64/openvr_api.lib" ws2_32 user32 gdi32)
    add_custom_command(TARGET vrst-sender POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${VRST_OPENVR_DIR}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:vrst-sender>/openvr_api.dll")
    install(FILES "${VRST_OPENVR_DIR}/bin/win64/openvr_api.dll" DESTINATION sender)
  endif()
  install(TARGETS vrst-sender RUNTIME DESTINATION sender)
  if(VRST_BUILD_TESTS)
    add_executable(vrst-sender-tests sender/sender_tests.cpp)
    target_link_libraries(vrst-sender-tests PRIVATE vrst_core)
    add_test(NAME sender-math COMMAND vrst-sender-tests)
    if(WIN32)
      add_executable(vrst-sender-overlay-tests sender/overlay_tests.cpp sender/overlay.cpp)
      target_link_libraries(vrst-sender-overlay-tests PRIVATE vrst_core user32 gdi32)
      add_test(NAME sender-overlay COMMAND vrst-sender-overlay-tests)
    endif()
    if(NOT CMAKE_CROSSCOMPILING)
      find_package(Python3 COMPONENTS Interpreter)
      if(Python3_FOUND)
        add_test(NAME sender-integration COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/sender/integration_tests.py" "$<TARGET_FILE:vrst-sender>")
        if(WIN32)
          add_executable(vrst-launcher-exit-fixture sender/launcher_exit_fixture.cpp)
          add_test(NAME sender-launchers COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/sender/launcher_tests.py" "$<TARGET_FILE:vrst-launcher-exit-fixture>")
        endif()
      endif()
    endif()
  endif()
endif()
if(MSVC)
  target_compile_options(vrst_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(vrst_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(VRST_BUILD_TESTS)
  enable_testing()
  foreach(suite advanced calibration mirror spectator)
    add_executable(vrst-${suite}-tests tests/${suite}_tests.cpp)
    target_link_libraries(vrst-${suite}-tests PRIVATE vrst_core)
    add_test(NAME ${suite} COMMAND vrst-${suite}-tests)
  endforeach()
  add_executable(vrst-calibration-session-tests tests/calibration_session_tests.cpp plugin/calibration_session.cpp)
  target_include_directories(vrst-calibration-session-tests PRIVATE plugin)
  target_link_libraries(vrst-calibration-session-tests PRIVATE vrst_core)
  add_test(NAME calibration-session COMMAND vrst-calibration-session-tests)
  add_executable(vrst-image-tests tests/image_tests.cpp)
  target_link_libraries(vrst-image-tests PRIVATE vrst_core)
  add_test(NAME image COMMAND vrst-image-tests)
  add_executable(vrst-timing-overlay-tests tests/timing_overlay_tests.cpp)
  target_link_libraries(vrst-timing-overlay-tests PRIVATE vrst_core)
  add_test(NAME timing-overlay COMMAND vrst-timing-overlay-tests)
  add_executable(vrst-core-tests tests/core_tests.cpp)
  target_link_libraries(vrst-core-tests PRIVATE vrst_core)
  add_test(NAME core COMMAND vrst-core-tests)
  add_executable(vrst-network-tests tests/network_tests.cpp)
  target_link_libraries(vrst-network-tests PRIVATE vrst_receiver)
  add_test(NAME network COMMAND vrst-network-tests)
endif()
